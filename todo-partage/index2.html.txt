const FB_PROJECT = 'familyhub-colis-8abbd';
const FB_COLL    = 'meta/todo/tasks';
function getFBKey() {
  try { const s = JSON.parse(localStorage.getItem('lt_fb')); return s && s.apiKey ? s.apiKey : ''; } catch { return ''; }
}
function fbUrl(path) {
  return 'https://firestore.googleapis.com/v1/projects/' + FB_PROJECT +
    '/databases/(default)/documents/' + path + '?key=' + getFBKey();
}
function toFields(obj) {
  const f = {};
  for (const k in obj) {
    const v = obj[k];
    if (v === null || v === undefined) f[k] = {nullValue: null};
    else if (typeof v === 'boolean')   f[k] = {booleanValue: v};
    else if (typeof v === 'number')    f[k] = {integerValue: String(v)};
    else                               f[k] = {stringValue: String(v)};
  }
  return f;
}
function fromFields(fields) {
  if (!fields) return {};
  const obj = {};
  for (const k in fields) {
    const v = fields[k];
    if ('booleanValue' in v)      obj[k] = v.booleanValue;
    else if ('integerValue' in v) obj[k] = parseInt(v.integerValue);
    else if ('stringValue' in v)  obj[k] = v.stringValue;
    else obj[k] = null;
  }
  return obj;
}
async function fbWrite(task) {
  if (!getFBKey()) return;
  try {
    await fetch(fbUrl(FB_COLL + '/' + task.id), {
      method: 'PATCH',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({fields: toFields(task)})
    });
  } catch(e) { console.warn('fbWrite', e); }
}
async function fbDeleteTask(id) {
  if (!getFBKey()) return;
  try { await fetch(fbUrl(FB_COLL + '/' + id), {method:'DELETE'}); }
  catch(e) { console.warn('fbDelete', e); }
}
async function fbLoadAll() {
  if (!getFBKey()) return null;
  try {
    const r = await fetch(fbUrl(FB_COLL));
    if (!r.ok) return null;
    const data = await r.json();
    return (data.documents || []).map(doc => fromFields(doc.fields));
  } catch(e) { console.warn('fbLoadAll', e); return null; }
}

function deadlineClass(iso) {
  if (!iso) return '';
  const diff = new Date(iso) - new Date().setHours(0,0,0,0);
  const days = Math.ceil(diff / 86400000);
  if (days < 0)   return 'urgent';
  if (days <= 2)  return 'urgent';
  if (days <= 7)  return 'soon';
  return 'ok';
}
function fmtDeadline(iso) {
  if (!iso) return '';
  const d = new Date(iso);
  const diff = new Date(iso) - new Date().setHours(0,0,0,0);
  const days = Math.ceil(diff / 86400000);
  if (days === 0) return '📅 Aujourd\'hui';
  if (days === 1) return '📅 Demain';
  if (days < 0)   return '⚠️ ' + d.toLocaleDateString('fr-FR', {day:'numeric',month:'short'});
  return '📅 ' + d.toLocaleDateString('fr-FR', {day:'numeric',month:'short'});
}
function fmtDate(iso) {
  if (!iso) return '';
  const d = new Date(iso), now = new Date();
  if (d.toDateString() === now.toDateString())
    return 'Aujourd\'hui ' + d.toLocaleTimeString('fr-FR', {hour:'2-digit', minute:'2-digit'});
  return d.toLocaleDateString('fr-FR', {day:'numeric', month:'short'});
}

function createCalEvent(id) {
  const t = D.tasks.find(x => x.id === id); if (!t) return;

  let url = 'https://calendar.google.com/calendar/render?action=TEMPLATE&text=' + title + '&details=' + desc;
  if (t.deadline) {
    const d  = t.deadline.replace(/-/g,'');
    const dt = new Date(t.deadline); dt.setDate(dt.getDate()+1);
    const de = dt.toISOString().slice(0,10).replace(/-/g,'');
    url += '&dates=' + d + '/' + de;
  }
  if (window.Capacitor && window.Capacitor.Plugins && window.Capacitor.Plugins.Browser) {
    window.Capacitor.Plugins.Browser.open({url});
  } else {
    window.open(url, '_blank');
  }
  showToast('📅 Agenda ouvert');
}

function getFilt() {
  let tasks = [...D.tasks];
  if      (flt === 'todo') tasks = tasks.filter(t => !t.done);
  else if (flt === 'done') tasks = tasks.filter(t =>  t.done);
  else if (flt === 'g')    tasks = tasks.filter(t => t.who === 'Guillaume' || t.who === 'Tous');
  else if (flt === 'm')    tasks = tasks.filter(t => t.who === 'Michele'   || t.who === 'Tous');
  return tasks.sort((a, b) => {
    if (a.done !== b.done) return a.done ? 1 : -1;
    if (a.deadline && b.deadline) return new Date(a.deadline) - new Date(b.deadline);
    if (a.deadline && !b.deadline) return -1;
    if (!a.deadline && b.deadline) return 1;
    const po = {high:0, mid:1, low:2};
    return (po[a.prio]||1) - (po[b.prio]||1);
  });
}

  t.textContent = m; t.classList.add('show');
  clearTimeout(_tt); _tt = setTimeout(() => t.classList.remove('show'), 3000);
}

/* FIX: findAction avec guard nodeType===1 (évite crash TextNode Capacitor) */
function findAction(elem) {
  var el = elem;
  while (el && el !== document) {
    if (el.nodeType === 1 && el.getAttribute && el.getAttribute('data-action')) return el;
    el = el.parentNode;
  }
  return null;
}

document.addEventListener('click', function(e) {
  const el = findAction(e.target); if (!el) return;
  const action = el.getAttribute('data-action');
  const ds = el.dataset;
  switch (action) {
    case 'toggleTheme': if(window.toggleTheme) window.toggleTheme(); break;
    case 'openAdd':     openAddSheet(); break;
    case 'closeAdd':    closeSheet('shAdd'); break;
    case 'setFilter':
      flt = ds.filter || 'all';
      document.querySelectorAll('.chip').forEach(b => b.classList.remove('sel'));
      el.classList.add('sel'); render(); break;
    case 'selPrio':
      prio = ds.prio;
      document.querySelectorAll('.pbtn').forEach(b => b.classList.remove('sel'));
      el.classList.add('sel'); break;
    case 'saveTask':   saveTask(); break;
    case 'toggleDone': toggleDone(parseInt(ds.id)); break;
    case 'editTask':
      e.stopPropagation();
      openEditSheet(parseInt(ds.id)); break;
    case 'delTask':
      e.stopPropagation();
      delTask(parseInt(ds.id)); break;
    case 'calTask':
      e.stopPropagation();
      createCalEvent(parseInt(ds.id)); break;
  }
});
document.addEventListener('click', function(e) {
  if (e.target.classList && e.target.classList.contains('sov')) closeSheet(e.target.id);
});
document.addEventListener('keydown', function(e) {

async function init() {
  document.querySelectorAll('.sov').forEach(el => { el.style.display = 'none'; });
  render();
  const remote = await fbLoadAll();
  if (remote && remote.length > 0) {
    const localMap = {};
    D.tasks.forEach(t => { localMap[t.id] = t; });
    remote.forEach(t => { localMap[t.id] = t; });
    D.tasks = Object.values(localMap).sort((a,b) => new Date(b.date||0) - new Date(a.date||0));
    D.nextId = Math.max(D.nextId, ...D.tasks.map(t => t.id + 1));
    saveD(D); render();
  }
}
